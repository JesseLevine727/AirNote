cmake_minimum_required(VERSION 3.20)
project(airnote_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(airnote_core firmware/src/instrument.cpp)
target_include_directories(airnote_core PUBLIC firmware/include)
target_compile_options(airnote_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(instrument_tests tests/instrument_tests.cpp)
target_link_libraries(instrument_tests PRIVATE airnote_core)
target_compile_options(instrument_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME instrument_behavior COMMAND instrument_tests)
add_executable(render_demo tools/render_demo.cpp)
target_link_libraries(render_demo PRIVATE airnote_core)
